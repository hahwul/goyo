{% component trans_or_default(key, default, lang, config, config_file) -%} {%- set effective_lang = lang | default(value=config.default_language | default(value="en")) -%} {# Load config data to safely check if translation keys exist before calling trans() #} {%- set loaded_config = config_file -%} {%- if not loaded_config -%} {%- set loaded_config = load_data(path="/config.toml", format="toml", required=false) -%} {%- if not loaded_config -%} {%- set loaded_config = load_data(path="/config.yaml", format="yaml", required=false) -%} {%- endif -%} {%- if not loaded_config -%} {%- set loaded_config = load_data(path="/config.yml", format="yaml", required=false) -%} {%- endif -%} {%- endif -%} {# Find a language that has the key before calling trans(), which errors on a missing key #} {%- set trans_lang = "" -%} {%- if loaded_config -%} {%- if effective_lang != config.default_language and loaded_config.languages and loaded_config.languages[effective_lang] and loaded_config.languages[effective_lang].translations and loaded_config.languages[effective_lang].translations[key] -%} {%- set trans_lang = effective_lang -%} {%- elif loaded_config.translations and loaded_config.translations[key] -%} {# The current language lacks the key: fall back to the default language's text #} {%- set trans_lang = config.default_language -%} {%- endif -%} {%- endif -%} {%- if trans_lang -%} {{ trans(key=key, lang=trans_lang) }} {%- else -%} {{ default }} {%- endif -%} {% endcomponent %}
